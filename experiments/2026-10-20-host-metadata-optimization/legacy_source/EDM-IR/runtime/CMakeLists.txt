cmake_minimum_required(VERSION 3.20)
project(edmir_runtime LANGUAGES CXX)

option(EDMIR_ENABLE_CUDA "Build the CUDA/NCCL/cuBLAS executor" OFF)
option(EDMIR_ENABLE_CUTLASS "Build the optional CUTLASS FP16 SM80 GEMM provider" OFF)
option(EDMIR_ENABLE_NCCL_DEVICE "Build the NCCL Device API fused reference kernel" OFF)
option(EDMIR_BUILD_TESTS "Build host runtime tests" ON)
set(EDMIR_CUTLASS_DIR "" CACHE PATH "Path to a CUTLASS source checkout")
set(EDMIR_EXPERIMENT_DIR "" CACHE PATH "Prepared selector experiment package")
set(
  EDMIR_GENERATED_DIR "" CACHE PATH
  "Directory containing an edmir_program.hpp generated by the C++ backend"
)

if(EDMIR_ENABLE_CUTLASS AND NOT EDMIR_ENABLE_CUDA)
  message(FATAL_ERROR "EDMIR_ENABLE_CUTLASS requires EDMIR_ENABLE_CUDA=ON")
endif()
if(EDMIR_ENABLE_NCCL_DEVICE AND NOT EDMIR_ENABLE_CUDA)
  message(FATAL_ERROR "EDMIR_ENABLE_NCCL_DEVICE requires EDMIR_ENABLE_CUDA=ON")
endif()

add_library(edmir_runtime STATIC src/validate.cpp src/uniform_workload.cpp)
target_include_directories(
  edmir_runtime
  PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:include>
)
target_compile_features(edmir_runtime PUBLIC cxx_std_20)

if(EDMIR_ENABLE_CUDA)
  find_package(CUDAToolkit REQUIRED)
  find_path(NCCL_INCLUDE_DIR nccl.h REQUIRED)
  find_library(NCCL_LIBRARY nccl REQUIRED)

  add_library(edmir_cuda_runtime STATIC src/cuda_backend.cpp src/merged_host_backend.cpp)
  target_include_directories(edmir_cuda_runtime PUBLIC ${NCCL_INCLUDE_DIR})
  target_link_libraries(
    edmir_cuda_runtime
    PUBLIC
      edmir_runtime
      CUDA::cudart
      CUDA::cublas
      CUDA::cublasLt
      ${NCCL_LIBRARY}
  )
  target_compile_features(edmir_cuda_runtime PUBLIC cxx_std_20)

  if(EDMIR_ENABLE_CUTLASS OR EDMIR_ENABLE_NCCL_DEVICE)
    enable_language(CUDA)
    set_property(TARGET edmir_cuda_runtime PROPERTY CUDA_ARCHITECTURES 80)
    set_property(TARGET edmir_cuda_runtime PROPERTY CUDA_STANDARD 20)
  endif()

  if(EDMIR_ENABLE_NCCL_DEVICE)
    if(NOT EXISTS "${NCCL_INCLUDE_DIR}/nccl_device.h")
      message(FATAL_ERROR "EDMIR_ENABLE_NCCL_DEVICE requires NCCL 2.28+ nccl_device.h")
    endif()
    target_sources(edmir_cuda_runtime PRIVATE src/nccl_device_fused.cu)
    target_compile_definitions(edmir_cuda_runtime PRIVATE EDMIR_HAS_NCCL_DEVICE=1)
  endif()

  if(EDMIR_ENABLE_CUTLASS)
    if(NOT EDMIR_CUTLASS_DIR)
      message(FATAL_ERROR "EDMIR_ENABLE_CUTLASS requires EDMIR_CUTLASS_DIR")
    endif()
    if(NOT EXISTS "${EDMIR_CUTLASS_DIR}/include/cutlass/cutlass.h")
      message(FATAL_ERROR "EDMIR_CUTLASS_DIR is not a CUTLASS source checkout")
    endif()
    target_sources(edmir_cuda_runtime PRIVATE src/cutlass_gemm.cu)
    target_include_directories(edmir_cuda_runtime PRIVATE "${EDMIR_CUTLASS_DIR}/include")
    target_compile_definitions(edmir_cuda_runtime PRIVATE EDMIR_HAS_CUTLASS=1)
    target_compile_options(
      edmir_cuda_runtime PRIVATE
      $<$<COMPILE_LANGUAGE:CUDA>:--expt-relaxed-constexpr>
    )

    if(EDMIR_ENABLE_NCCL_DEVICE)
      target_sources(edmir_cuda_runtime PRIVATE src/nccl_device_fused_cutlass.cu)
      target_compile_definitions(
        edmir_cuda_runtime PRIVATE EDMIR_HAS_NCCL_DEVICE_CUTLASS=1
      )
    endif()

    add_executable(edmir_benchmark_cutlass_gemm tools/benchmark_cutlass_gemm.cpp)
    target_link_libraries(edmir_benchmark_cutlass_gemm PRIVATE edmir_cuda_runtime)
    target_compile_features(edmir_benchmark_cutlass_gemm PRIVATE cxx_std_20)
  endif()

  add_executable(edmir_benchmark_gemm tools/benchmark_gemm.cpp)
  target_link_libraries(edmir_benchmark_gemm PRIVATE CUDA::cudart CUDA::cublas)
  target_compile_features(edmir_benchmark_gemm PRIVATE cxx_std_20)

  add_executable(edmir_benchmark_copy tools/benchmark_copy.cpp)
  target_link_libraries(edmir_benchmark_copy PRIVATE CUDA::cudart)
  target_compile_features(edmir_benchmark_copy PRIVATE cxx_std_20)

  if(EDMIR_GENERATED_DIR)
    if(NOT EXISTS "${EDMIR_GENERATED_DIR}/edmir_program.hpp")
      message(FATAL_ERROR "EDMIR_GENERATED_DIR does not contain edmir_program.hpp")
    endif()
    add_executable(edmir_run_generated tools/run_generated.cpp)
    target_include_directories(edmir_run_generated PRIVATE "${EDMIR_GENERATED_DIR}")
    target_link_libraries(edmir_run_generated PRIVATE edmir_cuda_runtime)
    target_compile_features(edmir_run_generated PRIVATE cxx_std_20)

    add_executable(edmir_validate_generated tools/validate_generated.cpp)
    target_include_directories(edmir_validate_generated PRIVATE "${EDMIR_GENERATED_DIR}")
    target_link_libraries(edmir_validate_generated PRIVATE edmir_cuda_runtime)
    target_compile_features(edmir_validate_generated PRIVATE cxx_std_20)

    add_executable(edmir_benchmark_generated tools/benchmark_generated.cpp)
    target_include_directories(edmir_benchmark_generated PRIVATE "${EDMIR_GENERATED_DIR}")
    target_link_libraries(edmir_benchmark_generated PRIVATE edmir_cuda_runtime)
    target_compile_features(edmir_benchmark_generated PRIVATE cxx_std_20)
  endif()

endif()

if(EDMIR_EXPERIMENT_DIR)
  if(EDMIR_ENABLE_CUDA AND (NOT EDMIR_ENABLE_CUTLASS OR NOT EDMIR_ENABLE_NCCL_DEVICE))
    message(FATAL_ERROR "Selector GPU experiments require both CUTLASS and NCCL Device API")
  endif()
  include(CTest)
  include("${EDMIR_EXPERIMENT_DIR}/generated_targets.cmake")
endif()

if(EDMIR_BUILD_TESTS)
  include(CTest)
  add_executable(edmir_runtime_program_test tests/program_test.cpp)
  target_link_libraries(edmir_runtime_program_test PRIVATE edmir_runtime)
  add_test(NAME edmir_runtime_program_test COMMAND edmir_runtime_program_test)
endif()
