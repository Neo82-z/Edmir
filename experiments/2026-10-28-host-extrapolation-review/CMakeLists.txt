cmake_minimum_required(VERSION 3.30)
project(edmir_host_phase_diagnostic LANGUAGES CXX)
get_filename_component(EDMIR_REPO "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
include(CTest)
add_executable(host_phase_timing_test "${EDMIR_REPO}/EDM-IR/runtime/tests/host_phase_timing_test.cpp")
target_include_directories(host_phase_timing_test PRIVATE "${EDMIR_REPO}/EDM-IR/runtime/include")
target_compile_features(host_phase_timing_test PRIVATE cxx_std_20)
add_test(NAME host_phase_timing COMMAND host_phase_timing_test)
set_tests_properties(host_phase_timing PROPERTIES TIMEOUT 10)
option(EDMIR_PHASE_CUDA "Build separate GPU diagnostic frontend" OFF)
if(EDMIR_PHASE_CUDA)
  if(NOT EXISTS "${EDMIR_PHASE_PROGRAM}/edmir_program.hpp")
    message(FATAL_ERROR "EDMIR_PHASE_PROGRAM must be an existing generated per-source program directory")
  endif()
  set(EDMIR_ENABLE_CUDA ON CACHE BOOL "" FORCE)
  set(EDMIR_ENABLE_CUTLASS ON CACHE BOOL "" FORCE)
  set(EDMIR_ENABLE_NCCL_DEVICE ON CACHE BOOL "" FORCE)
  add_subdirectory("${EDMIR_REPO}/EDM-IR/runtime" runtime)
  add_executable(edmir_diagnose_phases "${EDMIR_REPO}/EDM-IR/runtime/tools/diagnose_selector_phases.cpp")
  target_include_directories(edmir_diagnose_phases PRIVATE "${EDMIR_PHASE_PROGRAM}")
  target_link_libraries(edmir_diagnose_phases PRIVATE edmir_cuda_runtime)
  target_compile_features(edmir_diagnose_phases PRIVATE cxx_std_20)
endif()
